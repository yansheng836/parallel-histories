## Task 3: JS 重写布局函数（核心）

**Files:**

- Modify: `index.html` `<script>` 内「2) 分期色带定位」段（现 579–599 行：从 `// 2) 分期色带定位：…` 到 `layoutBands();` 全部替换）

- [ ] **Step 1: 整段替换为 `layoutTimeline()`**

删除原 `const tl=…` 到 `layoutBands();` 的整段，替换为：

```js
// 2) 真实时间比例尺（桌面端）：刻度/内容/色带按 data-date 等比铺开，内容放不下的区间局部膨胀
const tl=document.querySelector('.timeline');
const TICK_GAP=12;   // 刻度胶囊与内容、内容相互之间的最小间距
const K_MIN=0.25;    // 比例密度下限（px/年），防止极端窄屏时先秦段被压没

function layoutTimeline(){
  const isMobile=!window.matchMedia('(min-width:769px)').matches;
  const bands=[...tl.querySelectorAll('.era-cn,.era-west')];
  const breaks=[...tl.querySelectorAll('.axis-break')];

  // 手机端：清除全部内联定位，回到文档流单列（现状布局）
  if(isMobile){
    tl.querySelectorAll('.tick,.row,.link-note').forEach(el=>{
      el.style.position='';el.style.left='';el.style.right='';
      el.style.top='';el.style.margin='';
    });
    bands.forEach(b=>{b.style.top='';b.style.height='';});
    breaks.forEach(b=>b.remove());
    tl.style.height='';
    return;
  }

  // 按 DOM 顺序把 row/link-note 归属到其前面最近的刻度，收集 data-date
  const groups=[];let cur=null;
  for(const el of tl.children){
    if(el.classList.contains('axis')||el.classList.contains('era-cn')
      ||el.classList.contains('era-west')||el.classList.contains('axis-break'))continue;
    if(el.classList.contains('tick')){
      const d=parseFloat(el.dataset.date);
      if(isNaN(d)){console.warn('[timeline] 刻度缺少合法 data-date，比例尺布局中止：',el.id);return;}
      cur={el,date:d,items:[]};groups.push(cur);
    }else if(cur){cur.items.push(el);}
  }
  if(groups.length<2)return;
  for(let i=1;i<groups.length;i++){
    if(groups[i].date<=groups[i-1].date){
      console.warn('[timeline] 刻度 data-date 未按时间递增：',groups[i].el.id);return;
    }
  }

  const tickH=groups[0].el.offsetHeight||28;
  const minDate=groups[0].date;
  const axisEnd=Math.max(groups[groups.length-1].date,new Date().getFullYear());

  // Pass 1：每区间内容需求 need 与等比理想高度，最挤区间反推比例密度 k
  const segs=[];
  for(let i=0;i<groups.length-1;i++){
    const a=groups[i],b=groups[i+1];
    const span=b.date-a.date;
    const contentH=a.items.reduce((s,el)=>s+el.offsetHeight,0);
    const need=tickH+contentH+TICK_GAP*(a.items.length+1);
    segs.push({a,b,span,need});
  }
  let k=K_MIN;
  for(const s of segs)k=Math.max(k,s.need/s.span);
  const tailIdeal=(axisEnd-groups[groups.length-1].date)*k;

  // Pass 2：累加实际高度（膨胀段取 need），写内联 absolute 定位
  let y=0;
  segs.forEach(s=>{
    s.inflated=s.need>s.span*k;
    s.actual=s.inflated?s.need:s.span*k;
    s.y0=y;y+=s.actual;s.y1=y;
  });
  const H=y+tailIdeal+tickH;
  tl.style.height=H+'px';

  segs.forEach(s=>{
    s.a.el.style.position='absolute';
    s.a.el.style.left='0';s.a.el.style.right='0';s.a.el.style.top=s.y0+'px';
    let iy=s.y0+tickH+TICK_GAP;
    for(const it of s.a.items){
      it.style.position='absolute';
      it.style.left='0';it.style.right='0';
      it.style.top=iy+'px';it.style.margin='0';
      iy+=it.offsetHeight+TICK_GAP;
    }
  });
  const lastG=groups[groups.length-1];
  lastG.el.style.position='absolute';
  lastG.el.style.left='0';lastG.el.style.right='0';
  lastG.el.style.top=segs[segs.length-1].y1+'px';

  // date→y 映射（分段线性：非膨胀段斜率 k，膨胀段斜率=need/span；轴尾按 k 延伸）
  function yOfDate(d){
    if(d<=minDate)return 0;
    for(const s of segs){
      if(d<s.b.date)return s.y0+(d-s.a.date)/s.span*(s.y1-s.y0);
    }
    const lastSeg=segs[segs.length-1];
    return lastSeg.y1+(d-lastSeg.b.date)*k;
  }

  // 色带：直接用自身 data-start/data-end 计算（缺 start 时告警并跳过，不静默失败）
  bands.forEach(b=>{
    const start=parseFloat(b.dataset.start),end=parseFloat(b.dataset.end);
    if(isNaN(start)){console.warn('[timeline] 色带缺少合法 data-start：',b.textContent.trim().slice(0,12));return;}
    const top=yOfDate(start);
    const bottom=isNaN(end)?H:yOfDate(end);
    b.style.top=top+'px';
    b.style.height=Math.max(60,bottom-top-8)+'px';
  });

  // 断轴符号：仅在硬溢出（need > 等比高度）的区间中点插入
  breaks.forEach(b=>b.remove());
  segs.forEach(s=>{
    if(!s.inflated)return;
    const br=document.createElement('div');
    br.className='axis-break';
    br.style.top=(s.y0+s.actual/2)+'px';
    br.textContent='⧗ 此段比例已放大';
    br.title='该时间段内容较密，纵向按内容撑开，未严格按年等比';
    tl.appendChild(br);
  });
}
window.addEventListener('load',layoutTimeline);
window.addEventListener('resize',layoutTimeline);
if(document.fonts&&document.fonts.ready)document.fonts.ready.then(layoutTimeline);
if(window.ResizeObserver){new ResizeObserver(layoutTimeline).observe(tl);}
layoutTimeline();
```
